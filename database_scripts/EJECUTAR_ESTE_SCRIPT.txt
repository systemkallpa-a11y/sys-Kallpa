CONSULTA DE VERIFICACIÓN:

SELECT 
    id_presupuesto,
    numero_presupuesto,
    monto AS costos_directos,
    gastos_generales,
    utilidad,
    supervision_obra,
    igv,
    monto_total,
    -- Verificar cálculo correcto
    ROUND((monto + gastos_generales + utilidad) * 0.18, 2) AS igv_esperado,
    -- Comparar
    CASE 
        WHEN igv = ROUND((monto + gastos_generales + utilidad) * 0.18, 2) 
        THEN '✓ CORRECTO' 
        ELSE '❌ INCORRECTO' 
    END AS estado_calculo
FROM TblPresupuesto
ORDER BY id_presupuesto DESC
LIMIT 5;
